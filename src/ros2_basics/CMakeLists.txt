cmake_minimum_required(VERSION 3.8)
project(ros2_basics)
set(CMAKE_CXX_STANDARD 17)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(example_interfaces REQUIRED)
find_package(cell_interfaces REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(control_msgs REQUIRED)

add_executable(photo_eye src/photo_eye.cpp)
ament_target_dependencies(photo_eye rclcpp std_msgs)
add_executable(suction src/suction.cpp)
ament_target_dependencies(suction rclcpp std_srvs)
add_executable(countdown src/countdown.cpp)
ament_target_dependencies(countdown rclcpp rclcpp_action example_interfaces)
add_executable(box_pub src/box_pub.cpp)
ament_target_dependencies(box_pub rclcpp cell_interfaces)
add_executable(tf_demo src/tf_demo.cpp)
ament_target_dependencies(tf_demo rclcpp tf2_ros geometry_msgs)
add_executable(traj_client src/traj_client.cpp)
ament_target_dependencies(traj_client rclcpp rclcpp_action control_msgs)

install(TARGETS photo_eye suction countdown box_pub tf_demo traj_client DESTINATION lib/${PROJECT_NAME})
ament_package()
