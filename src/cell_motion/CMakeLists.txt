cmake_minimum_required(VERSION 3.8)
project(cell_motion)
set(CMAKE_CXX_STANDARD 17)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(cell_planning REQUIRED)
find_package(cell_interfaces REQUIRED)
find_package(std_srvs REQUIRED)
find_package(vision_msgs REQUIRED)
find_package(shape_msgs REQUIRED)

add_executable(motion_demo src/motion_demo.cpp)
target_include_directories(motion_demo PUBLIC include)
ament_target_dependencies(motion_demo rclcpp moveit_ros_planning_interface moveit_msgs tf2_ros tf2_eigen geometry_msgs)
target_link_libraries(motion_demo cell_planning::cell_planning)

add_executable(grip_demo src/grip_demo.cpp)
target_include_directories(grip_demo PUBLIC include)
ament_target_dependencies(grip_demo rclcpp moveit_ros_planning_interface moveit_msgs tf2_ros tf2_eigen geometry_msgs cell_interfaces)
target_link_libraries(grip_demo cell_planning::cell_planning)

add_executable(pick_place_server src/pick_place_server.cpp)
target_include_directories(pick_place_server PUBLIC include)
ament_target_dependencies(pick_place_server rclcpp rclcpp_action moveit_ros_planning_interface moveit_msgs shape_msgs tf2_ros tf2_eigen geometry_msgs cell_interfaces)
target_link_libraries(pick_place_server cell_planning::cell_planning)

add_executable(pick_demo src/pick_demo.cpp)
target_include_directories(pick_demo PUBLIC include)
ament_target_dependencies(pick_demo rclcpp rclcpp_action tf2_ros tf2_eigen geometry_msgs std_srvs vision_msgs cell_interfaces)
target_link_libraries(pick_demo cell_planning::cell_planning)

install(TARGETS motion_demo grip_demo pick_place_server pick_demo DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_package()
