cmake_minimum_required(VERSION 3.8)
project(cell_perception)
set(CMAKE_CXX_STANDARD 17)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(vision_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(cell_planning REQUIRED)
find_package(cell_interfaces REQUIRED)

add_executable(box_detector src/box_detector.cpp)
target_include_directories(box_detector PUBLIC include)
ament_target_dependencies(box_detector rclcpp sensor_msgs vision_msgs visualization_msgs tf2_ros tf2_eigen)
target_link_libraries(box_detector cell_planning::cell_planning Eigen3::Eigen ${OpenCV_LIBS})
add_executable(eval_detector src/eval_detector.cpp)
ament_target_dependencies(eval_detector rclcpp vision_msgs tf2_ros tf2_eigen cell_interfaces)
target_link_libraries(eval_detector cell_planning::cell_planning Eigen3::Eigen)
install(TARGETS box_detector eval_detector DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_detect test/test_detect.cpp)
  target_include_directories(test_detect PUBLIC include)
  target_link_libraries(test_detect Eigen3::Eigen ${OpenCV_LIBS})
endif()
ament_package()
