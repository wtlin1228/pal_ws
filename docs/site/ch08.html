Ch 8 — The digital twin: world model, boxes, mock suction gripper (week 5)

Goal: a world_model node that holds the truth of the cell (every box, where it is, who holds it) and a mock_gripper that behaves like a vacuum ejector. At the end you pick a box from the pallet and drop it on the conveyor, with the robot of Ch 7.

Truth versus belief

The twin has two sides. world_model is truth; only mock devices may read it. The robot software (executive, planner, motion) may only know what the mock camera and mock sensors tell it: belief. If you ever import world_model state into the executive, the tutorial stops teaching you anything.

Node Owns Interfaces
world_model boxes, held box, mock gravity world/spawn_box, world/get_boxes, world/reset, world/grip (SetBool), topics world/boxes, world/markers, TF box_<id>
mock_gripper vacuum state, response delay gripper/set_suction (SetSuction), topic gripper/vacuum (Bool)
spawn_scene nothing; a CLI fills the pallet with a named scene

Language note: the twin is the one part of this tutorial that stays in Python. It mocks devices you throw away on a real cell, Python is quicker for that, and none of it runs on the robot side. Everything that talks to it from the robot side, starting with grip_demo below, is C++.

cell_sim/cell_sim/world_model.py

import copy, numpy as np, rclpy
from rclpy.node import Node
from rclpy.callback_groups import ReentrantCallbackGroup
from rclpy.executors import MultiThreadedExecutor
from rclpy.duration import Duration
from std_srvs.srv import Trigger, SetBool
from geometry_msgs.msg import TransformStamped, Vector3
from visualization_msgs.msg import Marker, MarkerArray
from tf2_ros import Buffer, TransformListener, TransformBroadcaster, TransformException
from cell_interfaces.msg import Box, BoxArray
from cell_interfaces.srv import SpawnBox, GetBoxes
from cell_sim.geometry import mat_from_pose, mat_from_tf, pose_from_mat, trans

PALLET   = dict(x=0.0,  y=0.75, z=0.144, sx=0.8, sy=0.6)   # deck top center and size, world frame
CONVEYOR = dict(x=-0.9, y=0.0,  z=0.70,  sx=0.5, sy=1.5)
SKU_COLORS = {'A': (0.91, 0.38, 0.15), 'B': (0.39, 0.43, 0.45), 'C': (0.82, 0.08, 0.10)}

class WorldModel(Node):
    """Single source of truth for the mocked cell. Mock devices read it; robot software must not."""

    def __init__(self):
        super().__init__('world_model')
        self.boxes: dict[str, Box] = {}          # id -> Box, poses always in 'world'
        self.held: str | None = None
        self.T_tcp_box = None                     # fixed offset while a box is held
        self.buf = Buffer(); self.tf_listener = TransformListener(self.buf, self)
        self.br = TransformBroadcaster(self)
        cb = ReentrantCallbackGroup()
        self.create_service(SpawnBox, 'world/spawn_box', self.spawn, callback_group=cb)
        self.create_service(GetBoxes, 'world/get_boxes', self.get_boxes, callback_group=cb)
        self.create_service(Trigger, 'world/reset', self.reset, callback_group=cb)
        self.create_service(SetBool, 'world/grip', self.grip, callback_group=cb)
        self.pub_boxes = self.create_publisher(BoxArray, 'world/boxes', 10)
        self.pub_markers = self.create_publisher(MarkerArray, 'world/markers', 10)
        self.create_timer(0.1, self.tick)

    # ---- services -----------------------------------------------------------------
    def spawn(self, req, res):
        box = copy.deepcopy(req.box)
        if box.pose.header.frame_id != 'world':          # store everything in one frame
            try:
                tf = self.buf.lookup_transform('world', box.pose.header.frame_id, rclpy.time.Time(),
                                               timeout=Duration(seconds=2.0))
            except TransformException as e:                 # TF not up yet: answer, never hang the caller
                res.success, res.message = False, f'no transform for {box.pose.header.frame_id}: {e}'; return res
            box.pose.pose = pose_from_mat(mat_from_tf(tf.transform) @ mat_from_pose(box.pose.pose))
            box.pose.header.frame_id = 'world'
        box.location = self.location_of(box)
        self.boxes[box.id] = box
        res.success, res.message = True, f'spawned {box.id} on {box.location}'
        return res

    def get_boxes(self, req, res):
        res.boxes = [b for b in self.boxes.values() if not req.location or b.location == req.location]
        return res

    def reset(self, req, res):
        self.boxes.clear(); self.held = None
        res.success = True; return res

    def grip(self, req, res):
        """on: attach the box whose top face is under the TCP. off: release it and let it settle."""
        try:
            T_world_tcp = mat_from_tf(self.buf.lookup_transform('world', 'tcp', rclpy.time.Time()).transform)
        except TransformException as e:
            res.success, res.message = False, str(e); return res
        if req.data:
            hit = self.box_under(T_world_tcp)
            if hit is None:
                res.success, res.message = False, 'no box under the pad'; return res
            self.held = hit
            self.T_tcp_box = np.linalg.inv(T_world_tcp) @ mat_from_pose(self.boxes[hit].pose.pose)
            self.boxes[hit].location = 'gripper'
            res.success, res.message = True, f'holding {hit}'
        else:
            if self.held:
                box, self.held = self.boxes[self.held], None
                self.settle(box)
            res.success, res.message = True, 'released'
        return res

    # ---- mock physics ---------------------------------------------------------------
    def box_under(self, T_world_tcp, tol=0.02):
        p = np.append(T_world_tcp[:3, 3], 1.0)
        for b in self.boxes.values():
            if b.location == 'gripper':
                continue
            lp = np.linalg.inv(mat_from_pose(b.pose.pose)) @ p     # TCP in the box's own frame
            if abs(lp[0]) <= b.size.x / 2 and abs(lp[1]) <= b.size.y / 2 and abs(lp[2] - b.size.z / 2) < tol:
                return b.id
        return None

    def settle(self, box):
        """Mock gravity: drop straight down onto the highest thing below (surface or box)."""
        T = mat_from_pose(box.pose.pose); x, y = T[0, 3], T[1, 3]
        support = 0.0
        for a in (PALLET, CONVEYOR):
            if abs(x - a['x']) <= a['sx'] / 2 and abs(y - a['y']) <= a['sy'] / 2:
                support = a['z']
        for o in self.boxes.values():                                # axis-aligned footprint overlap
            if o.id == box.id or o.location == 'gripper':
                continue
            To = mat_from_pose(o.pose.pose)
            if abs(To[0, 3] - x) < (o.size.x + box.size.x) / 2 and abs(To[1, 3] - y) < (o.size.y + box.size.y) / 2:
                support = max(support, To[2, 3] + o.size.z / 2)
        T[2, 3] = support + box.size.z / 2
        box.pose.pose = pose_from_mat(T)
        box.location = self.location_of(box)

    def location_of(self, box):
        x, y = box.pose.pose.position.x, box.pose.pose.position.y
        for name, a in (('pallet', PALLET), ('conveyor', CONVEYOR)):
            if abs(x - a['x']) <= a['sx'] / 2 and abs(y - a['y']) <= a['sy'] / 2:
                return name
        return 'unknown'

    # ---- 10 Hz: held box follows the TCP; publish truth for the mocks and RViz -------
    def tick(self):
        if self.held:
            try:
                T_world_tcp = mat_from_tf(self.buf.lookup_transform('world', 'tcp', rclpy.time.Time()).transform)
                self.boxes[self.held].pose.pose = pose_from_mat(T_world_tcp @ self.T_tcp_box)
            except TransformException:
                pass
        now = self.get_clock().now().to_msg()
        arr = BoxArray(); arr.header.stamp, arr.header.frame_id = now, 'world'
        markers = MarkerArray(); markers.markers.append(Marker(action=Marker.DELETEALL))
        for i, b in enumerate(self.boxes.values()):
            b.pose.header.stamp = now
            arr.boxes.append(b)
            m = Marker(type=Marker.CUBE, action=Marker.ADD, id=i, ns='boxes')
            m.header.frame_id, m.header.stamp = 'world', now
            m.pose, m.scale = b.pose.pose, b.size
            m.color.r, m.color.g, m.color.b = SKU_COLORS.get(b.sku, (0.6, 0.6, 0.6)); m.color.a = 0.9
            markers.markers.append(m)
            t = TransformStamped(); t.header.stamp, t.header.frame_id, t.child_frame_id = now, 'world', f'box_{b.id}'
            p = b.pose.pose.position
            t.transform.translation = Vector3(x=p.x, y=p.y, z=p.z); t.transform.rotation = b.pose.pose.orientation
            self.br.sendTransform(t)
        self.pub_boxes.publish(arr); self.pub_markers.publish(markers)

def main():
    rclpy.init(); n = WorldModel()
    ex = MultiThreadedExecutor(); ex.add_node(n); ex.spin()

cell_sim/cell_sim/mock_gripper.py

import time, rclpy
from rclpy.node import Node
from rclpy.callback_groups import ReentrantCallbackGroup
from rclpy.executors import MultiThreadedExecutor
from std_msgs.msg import Bool
from std_srvs.srv import SetBool
from cell_interfaces.srv import SetSuction

class MockGripper(Node):
    """Vacuum ejector + vacuum switch: a response delay, then a signal that is False if nothing sealed."""

    def __init__(self):
        super().__init__('mock_gripper')
        self.declare_parameter('vacuum_delay_s', 0.2)
        cb = ReentrantCallbackGroup()
        self.world = self.create_client(SetBool, 'world/grip', callback_group=cb)
        self.create_service(SetSuction, 'gripper/set_suction', self.handle, callback_group=cb)
        self.vacuum = False
        self.pub = self.create_publisher(Bool, 'gripper/vacuum', 10)
        self.create_timer(0.1, lambda: self.pub.publish(Bool(data=self.vacuum)))

    def handle(self, req, res):
        time.sleep(self.get_parameter('vacuum_delay_s').value)     # ejector response time
        r = self.world.call(SetBool.Request(data=req.on))         # blocking call: needs the multithreaded executor
        self.vacuum = bool(req.on and r.success)
        res.success, res.message = r.success, r.message
        return res

def main():
    rclpy.init(); n = MockGripper()
    ex = MultiThreadedExecutor(); ex.add_node(n); ex.spin()

cell_sim/cell_sim/spawn_scene.py

import argparse, rclpy
from rclpy.node import Node
from std_srvs.srv import Trigger
from geometry_msgs.msg import Vector3
from cell_interfaces.msg import Box
from cell_interfaces.srv import SpawnBox
from cell_sim.geometry import trans, pose_from_mat

SKUS = {'A': (0.30, 0.18, 0.15), 'B': (0.20, 0.20, 0.12), 'C': (0.40, 0.30, 0.20)}   # x, y, z [m]

def single_sku(layers):            # 2 x 3 grid of SKU A per layer with 2 cm gaps, column stacked (Ch 12 interlocks)
    return [('A', trans(x, y, 0.15 * L + 0.075))
            for L in range(layers) for x in (-0.16, 0.16) for y in (-0.2, 0.0, 0.2)]

def mixed():                       # three SKUs with gaps; two boxes sit on top of others
    return [('C', trans(-0.2, -0.15, 0.10)), ('B', trans(-0.2, 0.15, 0.06)), ('A', trans(0.2, -0.2, 0.075)),
            ('A', trans(0.2, 0.0, 0.075)), ('B', trans(0.2, 0.0, 0.21)), ('A', trans(-0.2, -0.15, 0.275))]

def call(node, client, req):
    fut = client.call_async(req); rclpy.spin_until_future_complete(node, fut); return fut.result()

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--scene', choices=['single_sku', 'mixed'], default='single_sku')
    ap.add_argument('--layers', type=int, default=2)
    args = ap.parse_args()
    rclpy.init(); node = Node('spawn_scene')
    reset, spawn = node.create_client(Trigger, 'world/reset'), node.create_client(SpawnBox, 'world/spawn_box')
    reset.wait_for_service(); spawn.wait_for_service()
    call(node, reset, Trigger.Request())
    scene = single_sku(args.layers) if args.scene == 'single_sku' else mixed()
    for n, (sku, T) in enumerate(scene):
        b = Box(id=f'box_{n:03d}', sku=sku, size=Vector3(x=SKUS[sku][0], y=SKUS[sku][1], z=SKUS[sku][2]))
        b.pose.header.frame_id = 'pallet_frame'; b.pose.pose = pose_from_mat(T)   # z = height above the deck
        print(call(node, spawn, SpawnBox.Request(box=b)).message)
    rclpy.shutdown()

First pick: cell_motion/src/grip_demo.cpp

This program reads the box pose from world/get_boxes. That is allowed only here, as a test of the twin; from Ch 10 on the pose comes from vision.

#include <cmath>
#include <thread>
#include <cell_interfaces/srv/get_boxes.hpp>
#include <cell_interfaces/srv/set_suction.hpp>
#include <cell_motion/robot.hpp>

using namespace cell_planning;
using GetBoxes = cell_interfaces::srv::GetBoxes;
using SetSuction = cell_interfaces::srv::SetSuction;

static geometry_msgs::msg::PoseStamped stamped(const Isometry3d & T, const std::string & frame = "world") {
  geometry_msgs::msg::PoseStamped p; p.header.frame_id = frame; p.pose = poseFromMat(T); return p;
}

int main(int argc, char ** argv) {
  rclcpp::init(argc, argv);
  auto node = rclcpp::Node::make_shared("grip_demo", rclcpp::NodeOptions().automatically_declare_parameters_from_overrides(true));
  rclcpp::executors::MultiThreadedExecutor ex;
  ex.add_node(node);
  std::thread spinner([&ex]() { ex.spin(); });
  cell_motion::Robot robot(node);
  auto get_boxes = node->create_client<GetBoxes>("world/get_boxes");
  auto suction = node->create_client<SetSuction>("gripper/set_suction");
  get_boxes->wait_for_service(); suction->wait_for_service();
  auto setSuction = [&](bool on) { auto r = std::make_shared<SetSuction::Request>(); r->on = on; return suction->async_send_request(r).get()->success; };

  auto req = std::make_shared<GetBoxes::Request>(); req->location = "pallet";
  const auto boxes = get_boxes->async_send_request(req).get()->boxes;          // test-only shortcut
  if (boxes.empty()) { RCLCPP_ERROR(node->get_logger(), "no boxes on the pallet"); rclcpp::shutdown(); spinner.join(); return 1; }
  const auto & box = boxes.front();
  const Isometry3d T_grasp = topDownTcp(matFromPose(box.pose.pose), box.size.z);  // TCP on the top face, pad down
  const Isometry3d T_above = T_grasp * trans(0, 0, -0.15);                         // 15 cm up (TCP z points down)
  const Isometry3d T_drop = trans(-0.9, 0.0, 0.70 + box.size.z) * rotX(M_PI);      // pad at box-top height on the belt

  bool ok = robot.goHome()
    && robot.moveTo(stamped(T_above)) && robot.moveLinear(stamped(T_grasp))
    && setSuction(true)
    && robot.moveLinear(stamped(T_above))
    && robot.moveTo(stamped(T_drop * trans(0, 0, -0.15))) && robot.moveLinear(stamped(T_drop));
  setSuction(false);
  ok = ok && robot.moveLinear(stamped(T_drop * trans(0, 0, -0.15))) && robot.goHome();
  RCLCPP_INFO(node->get_logger(), ok ? "done" : "a step failed");
  rclcpp::shutdown();
  spinner.join();
  return ok ? 0 : 1;
}
