Ch 7 — Programming motion: planners, Cartesian paths, IK (week 4)

Goal: a Robot class in cell_motion that moves the TCP to a pose, in joint space or in a straight line, from any C++ node. The pick-and-place server of Ch 11 is built on it.

What move_group exposes

Interface Type You use it for
/move_action action moveit_msgs/MoveGroup plan (and execute) to a pose or joint goal with a chosen pipeline and planner
/compute_cartesian_path service GetCartesianPath straight-line waypoints, returns the fraction achieved
/execute_trajectory action ExecuteTrajectory run a planned trajectory on the controller
/compute_ik, /compute_fk services reachability checks, seed-aware IK
/apply_planning_scene, /planning_scene service, topic add/remove/attach collision objects (Ch 11)

Any client library is a thin layer over these. In C++ you use MoveIt's own: moveit::planning_interface::MoveGroupInterface for goals, plans, execution and Cartesian paths, and PlanningSceneInterface for collision objects (Ch 11). Both come from moveit_ros_planning_interface.

#pragma once
#include <string>
#include <utility>
#include <vector>
#include <rclcpp/rclcpp.hpp>
#include <moveit/move_group_interface/move_group_interface.hpp>
#include <tf2_ros/buffer.h>
#include <tf2_ros/transform_listener.h>
#include <cell_planning/geometry.hpp>

namespace cell_motion {

/// Every motion the cell needs, expressed as TCP poses in `world`. One instance per node.
class Robot {
public:
  using MoveGroup = moveit::planning_interface::MoveGroupInterface;

  explicit Robot(const rclcpp::Node::SharedPtr & node)
  : node_(node), mg_(node, "arm"), buf_(node->get_clock()), listener_(buf_) {
    mg_.setEndEffectorLink("tcp");
    mg_.setPoseReferenceFrame("world");
    mg_.setMaxVelocityScalingFactor(0.5);           // scaling 0..1 of the joint_limits.yaml values
    mg_.setMaxAccelerationScalingFactor(0.5);
    mg_.setPlanningTime(2.0);
    mg_.setNumPlanningAttempts(5);
    setPlanner("ompl", "RRTConnectkConfigDefault");
  }

  void setPlanner(const std::string & pipeline, const std::string & planner) {
    pipeline_ = pipeline; planner_ = planner;
    mg_.setPlanningPipelineId(pipeline);            // "ompl" | "pilz_industrial_motion_planner"
    mg_.setPlannerId(planner);                      // "RRTConnectkConfigDefault" | "PTP" | "LIN"
  }

  bool goHome() { mg_.setNamedTarget("home"); return planAndExecute(); }

  /// Joint-space motion with the current planner; path shape is up to the planner.
  bool moveTo(const geometry_msgs::msg::PoseStamped & pose) { mg_.setPoseTarget(pose); return planAndExecute(); }

  /// Straight TCP line. Default: MoveIt's Cartesian-path service (IK per 5 mm step, collisions checked).
  /// use_pilz: the industrial LIN planner with the limits from pilz_cartesian_limits.yaml.
  bool moveLinear(const geometry_msgs::msg::PoseStamped & pose, bool use_pilz = false) {
    if (use_pilz) {
      const auto previous = std::make_pair(pipeline_, planner_);
      setPlanner("pilz_industrial_motion_planner", "LIN");
      const bool ok = moveTo(pose);
      setPlanner(previous.first, previous.second);
      return ok;
    }
    moveit_msgs::msg::RobotTrajectory traj;
    const double fraction = mg_.computeCartesianPath({pose.pose}, 0.005, traj, true);
    if (fraction < 0.95) {                          // refuse paths that stop short
      RCLCPP_WARN(node_->get_logger(), "cartesian path only %.0f %% feasible", 100 * fraction);
      return false;
    }
    return mg_.execute(traj) == moveit::core::MoveItErrorCode::SUCCESS;
  }

  std::vector<double> jointPositions() { return mg_.getCurrentJointValues(); }

  /// Ch 7 checkpoint: the TCP pose in world from TF, not from MoveIt.
  Eigen::Isometry3d tcpPose() {
    return cell_planning::matFromTf(
      buf_.lookupTransform("world", "tcp", tf2::TimePointZero, tf2::durationFromSec(1.0)).transform);
  }

  MoveGroup & mg() { return mg_; }

private:
  bool planAndExecute() {
    MoveGroup::Plan plan;
    if (mg_.plan(plan) != moveit::core::MoveItErrorCode::SUCCESS) return false;
    return mg_.execute(plan) == moveit::core::MoveItErrorCode::SUCCESS;
  }

  rclcpp::Node::SharedPtr node_;
  MoveGroup mg_;
  tf2_ros::Buffer buf_;
  tf2_ros::TransformListener listener_;
  std::string pipeline_, planner_;
};

}  // namespace cell_motion

A sequential program next to a spinning executor: src/motion_demo.cpp

#include <chrono>
#include <cmath>
#include <thread>
#include <cell_motion/robot.hpp>

using cell_planning::poseFromMat;
using cell_planning::rotX;
using cell_planning::trans;

geometry_msgs::msg::PoseStamped stamped(const Eigen::Isometry3d & T, const std::string & frame = "world") {
  geometry_msgs::msg::PoseStamped p;
  p.header.frame_id = frame;
  p.pose = poseFromMat(T);
  return p;
}

int main(int argc, char ** argv) {
  rclcpp::init(argc, argv);
  auto node = rclcpp::Node::make_shared("motion_demo",
                                        rclcpp::NodeOptions().automatically_declare_parameters_from_overrides(true));
  rclcpp::executors::MultiThreadedExecutor ex;
  ex.add_node(node);
  std::thread spinner([&ex]() { ex.spin(); });            // callbacks run here; main() stays a plain program

  cell_motion::Robot robot(node);                         // connects to move_group (robot model from the topic)
  const auto above = trans(0.0, 0.75, 0.144 + 0.45) * rotX(M_PI);   // 45 cm over the deck, pad facing down

  bool ok = robot.goHome();
  const auto t0 = std::chrono::steady_clock::now();
  ok = ok && robot.moveTo(stamped(above));                            // free-space move
  RCLCPP_INFO(node->get_logger(), "OMPL free-space move: %.2f s",
              std::chrono::duration<double>(std::chrono::steady_clock::now() - t0).count());
  ok = ok && robot.moveLinear(stamped(above * trans(0, 0, 0.15)));   // descend: +z in the TCP frame is DOWN
  ok = ok && robot.moveLinear(stamped(above));                       // retreat
  if (ok) {
    const double err = (robot.tcpPose().translation() - above.translation()).norm();
    RCLCPP_INFO(node->get_logger(), "TF vs goal: %.2f mm (the mock plugin is exact)", 1e3 * err);
    robot.setPlanner("pilz_industrial_motion_planner", "PTP");
    ok = robot.goHome();
  }
  RCLCPP_INFO(node->get_logger(), ok ? "done" : "a move failed");
  rclcpp::shutdown();
  spinner.join();
  return ok ? 0 : 1;
}

Under the hood: one raw service call

#include <moveit_msgs/srv/get_position_ik.hpp>

auto cli = node->create_client<moveit_msgs::srv::GetPositionIK>("/compute_ik");
cli->wait_for_service();
auto req = std::make_shared<moveit_msgs::srv::GetPositionIK::Request>();
req->ik_request.group_name = "arm";
req->ik_request.ik_link_name = "tcp";
req->ik_request.pose_stamped = stamped(above);
req->ik_request.avoid_collisions = true;
auto res = cli->async_send_request(req).get();    // blocks this thread; the executor thread serves the reply
RCLCPP_INFO(node->get_logger(), "error_code %d", res->error_code.val);   // 1 = SUCCESS, -31 = NO_IK_SOLUTION

This is all setPoseTarget + plan need from IK. The same pattern (moveit_msgs request → service → error code) covers every MoveIt interface.

Rules that save hours

  1. Express goals in world (or base_link) and say so with setPoseReferenceFrame; converting other frames yourself (Ch 4) removes a failure mode.
  2. Always check the return value. A failed plan leaves the robot where it was; the next relative move would then be wrong.
  3. Keep planning time short (2 s) and attempts > 1; retry with a slightly different approach pose before giving up (Ch 16).
  4. For descend/retreat use computeCartesianPath and refuse a fraction below 0.95; a 60 % path would stop the pad in mid-air.
