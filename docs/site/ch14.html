Ch 14 — Mock conveyor, I/O, palletizing end-to-end (week 9)

Goal: a mock_conveyor with a photo-eye and two PLC-style commands, and an executive that palletizes from the belt using the Ch 12 pattern. At the end you palletize twelve boxes, then depalletize them back onto the belt.

I/O, the way a cell sees it

Real device Signal Mock
Photo-eye at the pick point digital input, 1 = blocked /photo_eye (Bool, 20 Hz)
Barcode scanner / WMS message SKU of the box at the eye /conveyor/sku (String)
Outfeed belt "index" command PLC output + done signal conveyor/index_out (Trigger, returns when the eye clears)
Infeed belt "feed next" command PLC output + done signal conveyor/feed_in (Trigger, returns when the eye blocks)
Vacuum switch digital input /gripper/vacuum (Ch 8)

A PLC handshake is a service with a long response time: the request is the output you set, the response is the input you wait for.

cell_sim/cell_sim/mock_conveyor.py

The belt runs along +y of conveyor_frame, which is aligned with world, so y values are the same in both frames. world/spawn_box is an upsert: spawning an existing id moves it, which is how the belt moves boxes without a second interface. (Python, like the rest of the twin.)

import time, rclpy
from rclpy.node import Node
from rclpy.callback_groups import ReentrantCallbackGroup
from rclpy.executors import MultiThreadedExecutor
from std_msgs.msg import Bool, String
from std_srvs.srv import Trigger
from geometry_msgs.msg import Vector3
from cell_interfaces.msg import Box, BoxArray
from cell_interfaces.srv import SpawnBox
from cell_sim.geometry import trans, pose_from_mat
from cell_sim.spawn_scene import SKUS

EYE_Y, END_Y, START_Y = 0.0, 0.75, -0.70     # along the belt

class MockConveyor(Node):
    """A belt with a photo-eye at the robot's pick point and two PLC-style commands."""

    def __init__(self):
        super().__init__('mock_conveyor')
        self.declare_parameter('speed_mps', 0.2)
        self.declare_parameter('feed_sequence', 'AAAAAA')     # SKUs fed in order, repeating
        self.declare_parameter('jam', False)                  # Ch 14 checkpoint: a stuck belt
        cb = ReentrantCallbackGroup()
        self.boxes, self.running, self.fed = [], False, 0
        self.create_subscription(BoxArray, 'world/boxes', lambda m: setattr(self, 'boxes', m.boxes), 10, callback_group=cb)
        self.spawn = self.create_client(SpawnBox, 'world/spawn_box', callback_group=cb)
        self.pub_eye = self.create_publisher(Bool, 'photo_eye', 10)
        self.pub_sku = self.create_publisher(String, 'conveyor/sku', 10)
        self.create_service(Trigger, 'conveyor/index_out', self.index_out, callback_group=cb)
        self.create_service(Trigger, 'conveyor/feed_in', self.feed_in, callback_group=cb)
        self.create_timer(0.05, self.tick, callback_group=cb)

    def on_belt(self):
        return [b for b in self.boxes if b.location == 'conveyor']

    def at_eye(self):
        return [b for b in self.on_belt() if abs(b.pose.pose.position.y - EYE_Y) < b.size.y / 2]

    def tick(self):
        eye = self.at_eye()
        self.pub_eye.publish(Bool(data=bool(eye)))
        self.pub_sku.publish(String(data=eye[0].sku if eye else ''))
        if not self.running or self.get_parameter('jam').value:
            return
        dy = self.get_parameter('speed_mps').value * 0.05
        for b in self.on_belt():
            if b.pose.pose.position.y >= END_Y:                 # parked at the end: downstream takes over
                continue
            b.pose.pose.position.y += dy
            self.spawn.call_async(SpawnBox.Request(box=b))

    def index_out(self, req, res):
        """Run until the box at the eye has left it, plus half a second."""
        self.running, t0 = True, time.monotonic()
        while self.at_eye() and time.monotonic() - t0 < 10.0:
            time.sleep(0.05)
        time.sleep(0.5); self.running = False
        res.success, res.message = True, 'indexed'
        return res

    def feed_in(self, req, res):
        """Spawn the next box upstream and run until it blocks the eye."""
        seq = self.get_parameter('feed_sequence').value
        sku = seq[self.fed % len(seq)]; self.fed += 1
        b = Box(id=f'in_{self.fed:03d}', sku=sku, size=Vector3(x=SKUS[sku][0], y=SKUS[sku][1], z=SKUS[sku][2]))
        b.pose.header.frame_id = 'conveyor_frame'
        b.pose.pose = pose_from_mat(trans(0.0, START_Y, SKUS[sku][2] / 2))
        self.spawn.call(SpawnBox.Request(box=b))
        self.running, t0 = True, time.monotonic()
        while not self.at_eye() and time.monotonic() - t0 < 15.0:
            time.sleep(0.05)
        self.running = False
        res.success = bool(self.at_eye()); res.message = f'{sku} at eye' if res.success else 'feed timeout'
        return res

def main():
    rclpy.init(); n = MockConveyor()
    ex = MultiThreadedExecutor(); ex.add_node(n); ex.spin()

The pallet plan: cell_planning/include/cell_planning/palletizer.hpp

Both plans (single SKU here, mixed in Ch 15) share one small interface, so the executive does not care which one it holds.

#pragma once
#include <memory>
#include <optional>
#include <utility>
#include <vector>
#include <cell_planning/patterns.hpp>

namespace cell_planning {

using Placed = std::vector<std::pair<Isometry3d, BoxSize>>;     // belief: what sits on the pallet

/// What the executive needs from any placement plan: a target for the next box, confirmed once it is placed.
struct Plan {
  virtual ~Plan() = default;
  virtual std::optional<Isometry3d> target(const BoxSize & size) = 0;   // box center in pallet_frame, or nothing when full
  virtual void confirm() = 0;
  virtual const Placed & placed() const = 0;
};

/// Cursor over an interlocked single-SKU stack. target() is retried until confirm() advances it.
class SingleSkuPlan : public Plan {
public:
  explicit SingleSkuPlan(int layers = 3, const BoxSize & box = SKU_A) : box_(box) {
    int L = 0;
    for (const auto & layer : stack(layers)) {
      for (const auto & p : placeOrder(layer)) targets_.emplace_back(L, p);
      ++L;
    }
  }
  std::optional<Isometry3d> target(const BoxSize &) override {          // size ignored: every box is SKU_A
    if (i_ >= targets_.size()) return std::nullopt;
    return boxPose(targets_[i_].second, targets_[i_].first, box_);
  }
  void confirm() override { placed_.emplace_back(*target(box_), box_); ++i_; }
  const Placed & placed() const override { return placed_; }

private:
  BoxSize box_;
  std::vector<std::pair<int, Place>> targets_;
  std::size_t i_ = 0;
  Placed placed_;
};

}  // namespace cell_planning

The whole loop after these edits. Note the max_boxes check in FEED: the palletizing path never passes through DETECT, so without it max_boxes: 12 would be ignored.

  void execute(std::shared_ptr<JobHandle> gh) {
    auto fb = std::make_shared<RunJob::Feedback>();
    auto res = std::make_shared<RunJob::Result>();
    const auto & req = *gh->get_goal();
    const std::string mode = req.mode;
    std::string state = "CAPTURE";
    int moved = 0, retries = 0;
    Dets dets;
    std::optional<PickPlace::Goal> goal;
    if (mode == "palletize") { plan_ = std::make_unique<SingleSkuPlan>(3); state = "FEED"; }
    scene_.clear();
    while (true) {
      fb->state = state; fb->boxes_moved = moved; gh->publish_feedback(fb);
      RCLCPP_INFO(get_logger(), "[%d] %s", moved, state.c_str());
      if (gh->is_canceling()) { res->boxes_moved = moved; res->message = "cancelled"; gh->canceled(res); return; }

      if (state == "CAPTURE") {
        clearFrame();
        if (!capture()) { last_error_ = "camera timeout"; state = "RECOVER"; continue; }
        state = "DETECT";
      } else if (state == "DETECT") {
        const auto f = waitFrame(2s);
        if (!f) { last_error_ = "no detections"; state = "RECOVER"; continue; }
        dets = *f;
        if (dets.detections.empty() || (req.max_boxes >= 0 && moved >= req.max_boxes)) { state = "DONE"; continue; }
        state = "SELECT";
      } else if (state == "FEED") {
        if (req.max_boxes >= 0 && moved >= req.max_boxes) { state = "DONE"; continue; }
        const auto [ok, msg] = callTrigger(feed_in_, 30s);
        if (!ok) { last_error_ = msg; state = "RECOVER"; continue; }
        state = "SELECT";
      } else if (state == "SELECT") {
        goal = mode == "depalletize" ? planPick(dets) : planPlace();
        state = goal ? "PICK_PLACE" : "DONE";
      } else if (state == "PICK_PLACE") {
        const auto [ok, msg] = runPick(*goal);
        if (ok) {
          ++moved; retries = 0;
          if (mode == "depalletize") { callTrigger(index_out_, 15s); state = "CAPTURE"; }
          else { plan_->confirm(); state = "FEED"; }
        } else { last_error_ = msg; state = "RECOVER"; }
      } else if (state == "RECOVER") {
        ++retries;
        RCLCPP_ERROR(get_logger(), "recover (%d/3): %s", retries, last_error_.c_str());
        if (last_error_.find("still holding") != std::string::npos || retries > 3) { state = "FAILED"; continue; }
        state = mode == "depalletize" ? "CAPTURE" : "FEED";
      } else {                                                          // DONE or FAILED
        res->success = state == "DONE"; res->boxes_moved = moved; res->message = state;
        if (res->success) gh->succeed(res); else gh->abort(res);
        return;
      }
    }
  }

And the new method:

  /// Palletizing: the box at the eye goes to the next pattern slot.
  std::optional<PickPlace::Goal> planPlace() {
    std::string sku;
    { std::lock_guard<std::mutex> l(mtx_); if (!eye_ || !SKUS.count(sku_at_eye_)) return std::nullopt; sku = sku_at_eye_; }
    const BoxSize size = SKUS.at(sku);
    const auto T_target = plan_->target(size);
    if (!T_target) return std::nullopt;                                   // pallet full
    const Isometry3d T_belt_box = CONVEYOR_TOP * trans(0.0, 0.0, size.z / 2);   // guides center the box, yaw 0
    const auto [risk, T_pick] = bestGrasp(T_belt_box, size, {});
    const Isometry3d T_wp = tf2::transformToEigen(buf_.lookupTransform("world", "pallet_frame", tf2::TimePointZero, tf2::durationFromSec(1.0)));
    const Isometry3d T_place = T_wp * topDownTcp(*T_target, size.z);      // pad on the box top, pattern yaw
    scene_.clear();
    int n = 0;
    for (const auto & [T, s] : plan_->placed()) scene_.add("pal_" + std::to_string(n++), T, s);   // what we believe is on the pallet
    PickPlace::Goal g;
    g.box_id = "belt";
    g.pick_pose = stamped(T_pick);
    g.place_pose = stamped(T_place);
    g.box_size.x = size.x; g.box_size.y = size.y; g.box_size.z = size.z;
    return g;
  }

Pick and place use the same box-to-TCP relation (topDownTcp on both sides), so a pattern slot with yaw 90° simply turns the wrist during transit.

Run the full cycle

ros2 launch cell_bringup cell.launch.py
ros2 service call /world/reset std_srvs/srv/Trigger
ros2 action send_goal /run_job cell_interfaces/action/RunJob "{mode: palletize, strategy: single_sku, max_boxes: 12}" --feedback
# two interlocked layers now sit on the pallet; send them back:
ros2 action send_goal /run_job cell_interfaces/action/RunJob "{mode: depalletize, strategy: single_sku}" --feedback

Watch the feed: a box appears at y = −0.7, rides to the eye, /photo_eye flips to true, the arm takes it to its pattern slot, FEED starts the next. Depalletizing afterwards sees layer B on top (four rotated boxes) and handles it without any special case: the detector reports yaw, bestGrasp follows it.

Real cell: the belt is encoder-tracked and the PLC owns the handshakes; the robot controller receives "box present" and "SKU" and answers "box removed". Boxes on infeed belts are rarely centered, so most cells either add mechanical guides, a second camera over the belt, or measure the box in the gripper (Ch 18).
