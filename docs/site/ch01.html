Ch 1 — ROS 2 core: nodes, topics, services, actions, parameters (week 1)

Goal: write every kind of ROS 2 endpoint once, by hand, in a throwaway C++ package, so later chapters can use them without explanation.

The four interaction types

Type Pattern Use in our cell Example type
Topic many-to-many stream, no reply joint states, point clouds, photo-eye sensor_msgs/PointCloud2
Service request → reply, short suction on/off, reset world, trigger capture std_srvs/SetBool
Action goal → feedback stream → result, cancellable a pick-and-place, a whole job cell_interfaces/PickPlace
Parameter per-node config, changeable at runtime box size, camera noise, speed scale double, string[]

Rule of thumb: if it takes longer than a second or you may want to cancel it, it is an action.

Publisher with a parameter: src/photo_eye.cpp

#include <chrono>
#include <rclcpp/rclcpp.hpp>
#include <std_msgs/msg/bool.hpp>

class PhotoEye : public rclcpp::Node {
public:
  PhotoEye() : Node("photo_eye") {
    declare_parameter("period_s", 2.0);                    // how often a box "arrives"
    pub_ = create_publisher<std_msgs::msg::Bool>("photo_eye", 10);
    startTimer(get_parameter("period_s").as_double());
    // checkpoint: without this callback `ros2 param set` changes the stored value and nothing else
    param_cb_ = add_on_set_parameters_callback([this](const std::vector<rclcpp::Parameter> & params) {
      rcl_interfaces::msg::SetParametersResult result;
      result.successful = true;
      for (const auto & p : params) {
        if (p.get_name() == "period_s") {
          if (p.as_double() <= 0.0) { result.successful = false; result.reason = "period_s must be > 0"; return result; }
          startTimer(p.as_double());
        }
      }
      return result;
    });
  }

private:
  void startTimer(double period_s) {
    timer_ = create_wall_timer(std::chrono::duration<double>(period_s), [this]() { tick(); });
  }
  void tick() {
    blocked_ = !blocked_;
    std_msgs::msg::Bool msg;
    msg.data = blocked_;
    pub_->publish(msg);
    RCLCPP_INFO(get_logger(), "photo_eye blocked=%d", blocked_);
  }
  rclcpp::Publisher<std_msgs::msg::Bool>::SharedPtr pub_;
  rclcpp::TimerBase::SharedPtr timer_;
  rclcpp::node_interfaces::OnSetParametersCallbackHandle::SharedPtr param_cb_;
  bool blocked_{false};
};

int main(int argc, char ** argv) {
  rclcpp::init(argc, argv);
  rclcpp::spin(std::make_shared<PhotoEye>());            // runs timers and callbacks until Ctrl-C
  rclcpp::shutdown();
}

Service server: src/suction.cpp

#include <rclcpp/rclcpp.hpp>
#include <std_srvs/srv/set_bool.hpp>

class Suction : public rclcpp::Node {
public:
  Suction() : Node("suction") {
    srv_ = create_service<std_srvs::srv::SetBool>(
      "set_suction",
      [this](std_srvs::srv::SetBool::Request::ConstSharedPtr req, std_srvs::srv::SetBool::Response::SharedPtr res) {
        on_ = req->data;
        res->success = true;
        res->message = on_ ? "suction ON" : "suction OFF";
      });
  }

private:
  rclcpp::Service<std_srvs::srv::SetBool>::SharedPtr srv_;
  bool on_{false};
};

int main(int argc, char ** argv) {
  rclcpp::init(argc, argv);
  rclcpp::spin(std::make_shared<Suction>());
  rclcpp::shutdown();
}

Action server with feedback and cancel: src/countdown.cpp

#include <chrono>
#include <thread>
#include <rclcpp/rclcpp.hpp>
#include <rclcpp_action/rclcpp_action.hpp>
#include <example_interfaces/action/fibonacci.hpp>   // placeholder until Ch 2

using Fibonacci = example_interfaces::action::Fibonacci;
using GoalHandle = rclcpp_action::ServerGoalHandle<Fibonacci>;

class Countdown : public rclcpp::Node {
public:
  Countdown() : Node("countdown") {
    server_ = rclcpp_action::create_server<Fibonacci>(
      this, "countdown",
      [](const rclcpp_action::GoalUUID &, Fibonacci::Goal::ConstSharedPtr) { return rclcpp_action::GoalResponse::ACCEPT_AND_EXECUTE; },
      [](std::shared_ptr<GoalHandle>) { return rclcpp_action::CancelResponse::ACCEPT; },
      [this](std::shared_ptr<GoalHandle> gh) {
        std::thread([this, gh]() { execute(gh); }).detach();   // never block the executor with a long goal
      });
  }

private:
  void execute(std::shared_ptr<GoalHandle> gh) {
    const int n = gh->get_goal()->order;
    auto feedback = std::make_shared<Fibonacci::Feedback>();
    auto result = std::make_shared<Fibonacci::Result>();
    for (int i = 0; i < n; ++i) {
      if (gh->is_canceling()) { gh->canceled(result); return; }
      feedback->sequence = {n - i};
      gh->publish_feedback(feedback);
      std::this_thread::sleep_for(std::chrono::seconds(1));
    }
    result->sequence = {0};
    gh->succeed(result);
  }
  rclcpp_action::Server<Fibonacci>::SharedPtr server_;
};

int main(int argc, char ** argv) {
  rclcpp::init(argc, argv);
  rclcpp::spin(std::make_shared<Countdown>());           // feedback and cancel arrive while execute() sleeps in its thread
  rclcpp::shutdown();
}

Two gotchas you will hit later

  1. Waiting for a service or action result inside a callback of a single-threaded executor deadlocks: the reply needs the executor, which is busy running your callback. Fix: a MultiThreadedExecutor with a Reentrant callback group, or do the long work in its own std::thread as countdown does. The executive in Ch 13 is built on this.
  2. QoS must match. Sensor publishers (cameras) usually use best effort; a reliable subscriber will not receive them. Subscribe with rclcpp::SensorDataQoS() for point clouds and images.
